  <!-- 7. Homemade Burger Recipe -->
  <section id="recipe">
    <h1>Homemade Burger Recipe</h1>

    <h2>What You Need</h2>

    <ul>
      <li>1 burger bun</li>
      <li>1 beef patty</li>
      <li>1 slice of cheese</li>
      <li>Lettuce and tomato</li>
      <li>Onion</li>
      <li>Ketchup and mayonnaise</li>
      <li>Salt and pepper</li>
    </ul>

    <h2>Steps</h2>

    <ol>
      <li>Season the beef patty with salt and pepper.</li>
      <li>Heat a pan and cook the beef patty on both sides.</li>
      <li>
        <strong>Warning: make sure the beef is fully cooked</strong>
        before eating.
      </li>
      <li>Place the cheese on the patty and let it melt.</li>
      <li>Toast the burger bun and spread ketchup and mayonnaise on it.</li>
      <li>Add lettuce, tomato, and onion to the bun.</li>
      <li>
        Place the patty inside the bun and
        <em>enjoy your homemade burger!</em>
      </li>
    </ol>
  </section>
